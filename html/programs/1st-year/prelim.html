package com.crux.prelim1;

import java.util.Scanner;

public class Test1PreLim {

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);

        //Name
        String firstName, middleName, middleInitial, lastName, fullName;
        
        System.out.println("-----------NAME-----------");
        System.out.print("First Name: ");
        firstName = input.nextLine();
        System.out.print("Middle Initial: ");
        middleInitial = input.nextLine();
        System.out.print("Last Name: ");
        lastName = input.nextLine();
        fullName = lastName + ", " + firstName + " " + middleInitial + ". ";

        //PRELIM
        double pag, pcpg, pqg, peg;
        
        System.out.println("-------PRELIM-------");
        System.out.print("Attendance Grade: ");
        pag = input.nextDouble();
        System.out.print("Class Participation Grade: ");
        pcpg = input.nextDouble();
        System.out.print("Quiz Grade: ");
        peg = input.nextDouble();
        System.out.print("Exam Grade: ");
        pqg = input.nextDouble();

        //MIDTERM
        double mag, mcpg, mqg, meg;
        
        System.out.println("-------MIDTERM-------");
        System.out.print("Attendance Grade: ");
        mag = input.nextDouble();
        System.out.print("Class Participation Grade: ");
        mcpg = input.nextDouble();
        System.out.print("Quiz Grade: ");
        meg = input.nextDouble();
        System.out.print("Exam Grade: ");
        mqg = input.nextDouble();

        //FINAL
        double fag, fcpg, fqg, feg;
        
        System.out.println("-------FINAL-------");
        System.out.print("Attendance Grade: ");
        fag = input.nextDouble();
        System.out.print("Class Participation Grade: ");
        fcpg = input.nextDouble();
        System.out.print("Quiz Grade: ");
        feg = input.nextDouble();
        System.out.print("Exam Grade: ");
        fqg = input.nextDouble();

        //Summary
        double pGrade = (pag * 5 / 100) + (pcpg * 15 / 100) + (pqg * 30 / 100) + (peg * 50 / 100);
        double mGrade = (mag * 5 / 100) + (mcpg * 15 / 100) + (mqg * 30 / 100) + (meg * 50 / 100);
        double fGrade = (fag * 5 / 100) + (fcpg * 15 / 100) + (fqg * 30 / 100) + (feg * 50 / 100);
        double sGrade = (pGrade * 25 / 100) + (mGrade * 25 / 100) + (fGrade * 50 / 100);

        System.out.println("---------SUMMARY----------");
        System.out.println("Name: " + fullName);
        System.out.println("Prelim Grade: " + pGrade);
        System.out.println("Midterm Grade: " + mGrade);
        System.out.println("Final Grade: " + fGrade);

        if (sGrade >= 95 && sGrade <= 100) {
            System.out.println("Semester Grade: " + sGrade + " " + "(A)");
        } else if (sGrade >= 90 && sGrade < 95) {
            System.out.println("Semester Grade: " + sGrade + " " + "(B)");
        } else if (sGrade >= 85 && sGrade < 90) {
            System.out.println("Semester Grade: " + sGrade + " " + "(C)");
        } else if (sGrade >= 80 && sGrade < 85) {
            System.out.println("Semester Grade: " + sGrade + " " + "(D)");
        } else if (sGrade >= 75 && sGrade < 80) {
            System.out.println("Semester Grade: " + sGrade + " " + "(E)");
        } else {
            System.out.println("Semester Grade: " + sGrade + " " + "(F)");
        }
        if (sGrade >= 75 && sGrade < 100) {
            System.out.println("Remarks: PASSED");
        } else {
            System.out.println("Remarks: FAILED");
        }
    }
}
