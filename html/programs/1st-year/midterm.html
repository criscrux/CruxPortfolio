package com.crux.quiz1;

import java.util.Scanner;

public class Quiz1 {

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        double sum = 0;
        double difference = 0;
        double product = 1;
        String sumExpression = "";
        String diffExpression = "";
        String proExpression = "";

        System.out.print("Enter how many numbers you want to process: ");
        int count = input.nextInt();
        for (int i = 1; i <= count; i++) {
            System.out.print("Number: " + i + ": ");
            double number = input.nextDouble();
            sum = sum + number;
            product = product * number;
            if (i == 1) {
                sumExpression = sumExpression + number;
                diffExpression = diffExpression + number;
                proExpression = proExpression + number;
                difference = number;
            } else if (number < 0) {
                sumExpression = sumExpression + " - " + -number;
                diffExpression = diffExpression + " - " + "(" + number + ")";
                proExpression = proExpression + " * " + "(" + number + ")";
                difference = difference - number;
            } else {
                sumExpression = sumExpression + " + " + number;
                diffExpression = diffExpression + " - " + number;
                proExpression = proExpression + " * " + number;
                difference = difference - number;
            }
            if (i == count) {
                sumExpression = sumExpression + " = ";
                diffExpression = diffExpression + " = ";
                proExpression = proExpression + " = ";
            }
        }
        System.out.println("The inputted numbers are: ");
        System.out.println("Sum: " + sumExpression + sum);
        System.out.println("Difference: " + diffExpression + difference);
        System.out.println("Product: " + proExpression + product);
    }
}

package com.crux.quiz1;

import java.util.Scanner;

public class Quiz2 {

    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        System.out.println("Enter a Number:");
        int number = input.nextInt();

        for (int a = 1; a <= number; a++) {
            for (int b = 1; b <= number; b++) {
                if (a == 1 || a == number || b == 1 || b == number
                        || a == (number / 2 + 1)
                        || b == (number / 2 + 1)
                        || b == number - a + 1
                        //upper left
                        || a <= number / 2 + 1 && b <= a
                        //upper right
                        || a <= number / 2 + 1 && b > number / 2 + 1 && b <= number - a + 1
                        //down left
                        || a > number / 2 + 1 && b >= number - a + 1 && b <= number / 2 + 1
                        //down right
                        || a > number / 2 + 1 && b >= a
                        ) {
                    System.out.print("* ");
                } else {
                    System.out.print("  ");
                }
            }
            System.out.println();
        }
    }
}

package com.crux.act1;

import java.util.Scanner;

public class MidtermActivitysample2 {

    //Nested While Loop
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        System.out.println("Enter a number:");
        int number = input.nextInt();
        int rows = 2 * number - 1;

        int a = 1;
        while (a <= rows) {
            int diff = a - number;
            if (diff < 0) {
                diff = -diff;
            }

            //int indent = (number - 1) - diff;     // Right
            int indent = diff;                    // Left

            int i = 1;
            while (i <= number) {
                int c = 1;
                while (c <= 2 * number) {
                    if (c > indent && c <= indent + number) {
                        System.out.print("* ");
                    } else {
                        System.out.print("  ");
                    }
                    c++;
                }
                i++;
            }
            System.out.println();
            a++;
        }
    }
}

package com.crux.act1;

import java.util.Scanner;

public class MidtermActivitysample3 {

    //Nested While Loop
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        System.out.println("Enter a number:");
        int size = input.nextInt();
        int totalRows = 2 * size - 1;
 
        int currentRow = 1;
        while (currentRow <= totalRows) {
            int distanceFromMiddle = currentRow - size;
            if (distanceFromMiddle < 0) {
                distanceFromMiddle = -distanceFromMiddle;
            }

            int leadingSpaces = (size - 1) - distanceFromMiddle;     // Right
            //int leadingSpaces = distanceFromMiddle;                  // Left

            int patternCount = 1;
            while (patternCount <= size) {
                int currentCell = 1;
                while (currentCell <= 2 * size) {
                    if (currentCell > leadingSpaces && currentCell <= leadingSpaces + size) {
                        System.out.print("* ");
                    } else {
                        System.out.print("  ");
                    }
                    currentCell++;
                }
                patternCount++;
            }
            System.out.println();
            currentRow++;
        }
    }
}
